## 3. Auction Clearing (DA_60/DA_30/IDA1/IDA2)

```python
async def _clear_auction(self, auction_name: str):
    """Clear all eligible SPs in batch."""
    eligible_sps = self.current_phase["sp_range"]
    results = {}
    
    for sp in eligible_sps:
        # Freeze
        self.auction_locked.add(sp)
        
        # Aggregate curves
        supply = self._aggregate_supply(sp)
        demand = self._aggregate_demand(sp)
        
        # Uniform clearing price
        clearing_price, clearing_vol = self._uniform_clear(supply, demand)
        
        # Player allocations
        allocations = self._allocate_volumes(sp, clearing_price)
        
        results[sp] = {
            "auction": auction_name,
            "price": clearing_price,
            "volume": clearing_vol,
            "allocations": allocations
        }
        
        # Update positions
        self.positions[sp] = allocations
    
    await self.broadcast({"type": "auction_results", "results": results})

def _uniform_clear(self, supply: list, demand: list) -> tuple[float, float]:
    """Find marginal clearing price."""
    # Simplified: intersect cumulative curves
    supply.sort(key=lambda x: x[0])
    demand.sort(key=lambda x: x[0], reverse=True)
    
    s_cum, d_cum = 0, 0
    for s_price, s_vol in supply:
        s_cum += s_vol
        for d_price, d_vol in demand:
            d_cum += d_vol
            if s_price == d_price and abs(s_cum - d_cum) < 1:
                return s_price, min(s_cum, d_cum)
    
    return (supply[0][0] + demand[-1][0]) / 2, 0
```

## 4. Continuous ID (23 rounds, all SPs)

```python
async def id_tick(self):
    """Handle one continuous ID round."""
    round_num = (self.simtime_minute - 2070) // 30
    if round_num >= 23:  # Until BM
        return
        
    eligible_sps = range(self.current_sp, min(self.current_sp + 6, 49))
    
    for sp in eligible_sps:
        gc_time = self.gate_closure_for_sp(sp)
        if self.simtime_minute >= gc_time:
            continue
            
        # Match limit orders in order book
        matches = self._match_id_orders(sp)
        for match in matches:
            self.positions[sp][match["buyer"]] += match["mw"]
            self.positions[sp][match["seller"]] -= match["mw"]
    
    await self.broadcast({"type": "id_matches", "round": round_num, "matches": matches})
```

## 5. REALTIME/BM (all 48 SPs)

```python
async def _handle_realtime(self, old_simtime: int):
    """48 full SPs: GC → BM_OPEN → BM_CLEAR → SP_SETTLED."""
    sp_start_time = 2880 + (self.current_sp - 1) * 30
    
    if self.simtime_minute >= sp_start_time + 30:  # SP ended
        await self._settle_sp()
        self.current_sp += 1
        if self.current_sp > 48:
            await self._enter_results()
        else:
            self.bm_subphase = "BM_OPEN"
        return
    
    if self.bm_subphase == "BM_OPEN" and self.simtime_minute >= sp_start_time + 5:
        await self._run_bm_clear()
        self.bm_subphase = "BM_CLEAR"
    
    elif self.bm_subphase == "BM_CLEAR" and self.simtime_minute >= sp_start_time + 25:
        self.bm_subphase = "SP_SETTLED"

async def _run_bm_clear(self):
    """Full BM merit order + NIV + system price."""
    sp = self.current_sp
    
    # System imbalance
    demand = self._draw_demand(sp)
    res = self._draw_res(sp)
    system_imbalance = demand - res - sum(self.positions.get(sp, {}).values())
    
    # Stack offers/bids
    offers = self._get_bm_offers(sp)
    bids = self._get_bm_bids(sp)
    
    if system_imbalance > 0:  # Long → accept bids
        accepted = self._stack_bids(bids, system_imbalance)
        niv = -sum(v for _, v, _ in accepted)
    else:  # Short → accept offers
        accepted = self._stack_offers(offers, abs(system_imbalance))
        niv = sum(v for _, v, _ in accepted)
    
    system_price = self._marginal_price(accepted)
    
    await self.broadcast({
        "type": "bm_cleared",
        "sp": sp,
        "imbalance": system_imbalance,
        "niv": niv,
        "price": system_price,
        "accepted": accepted
    })

def gate_closure_for_sp(self, sp: int) -> int:
    """Exact GC time for each SP."""
    return 2880 + (sp - 1) * 30 - 30  # 1h before start
```
